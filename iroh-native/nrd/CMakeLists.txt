# libiroh_nrd: NVIDIA NRD v4.18.0 (stock, static, SPIR-V embedded) + the Iroh JNI shim in one shared library.
# NRD's own CMake downloads ShaderMake (zip, commit 3f01623a), MathLib (v11 zip) and a prebuilt DXC (v1.9.2609, GitHub release)
# at configure time: see knowledge/nrd.md for licences. Nothing here calls any graphics API.
cmake_minimum_required(VERSION 3.23)
project(iroh_nrd LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(IROH_NRD_SRC "${CMAKE_CURRENT_SOURCE_DIR}/nrd-src" CACHE PATH "fetched NRD source")
set(IROH_JNI_INCLUDE "" CACHE PATH "directory list with jni.h")
option(IROH_NRD_QUAD "NRD_SUPPORTS_QUAD_INTRINSICS (needs VK_KHR_compute_shader_derivatives at runtime)" ON)
set(NRD_STATIC_LIBRARY ON CACHE BOOL "" FORCE)
set(NRD_NRI OFF CACHE BOOL "" FORCE)
set(NRD_SUPPORTS_QUAD_INTRINSICS ${IROH_NRD_QUAD} CACHE BOOL "" FORCE)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/out")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/out")
# Cross builds (Windows): NRD's own CMake compiles its SPIR-V with a HOST dxc/ShaderMake at build time, which cannot run for
# a mingw target. The SPIR-V headers are platform independent, so the native Linux build (build-nrd.sh) generates them into
# <nrd-src>/_Shaders and a cross build compiles NRD's C++ directly against them (same flags, same sources, no NRD CMake).
set(IROH_NRD_MATHLIB_DIR "" CACHE PATH "MathLib source (the Linux build's _deps/mathlib-src); cross builds only")
set(IROH_NRD_SHADERMAKE_DIR "" CACHE PATH "ShaderMake source (the Linux build's _deps/shadermake-src); cross builds only")
if(IROH_NRD_MATHLIB_DIR)
  file(GLOB NRD_CXX ${IROH_NRD_SRC}/Source/*.cpp)
  if(NOT EXISTS ${IROH_NRD_SRC}/_Shaders/Clear.cs.spirv.h)
    message(FATAL_ERROR "no generated shaders in ${IROH_NRD_SRC}/_Shaders: run build-nrd.sh (Linux) first")
  endif()
  add_library(NRD STATIC ${NRD_CXX} ${IROH_NRD_SHADERMAKE_DIR}/ShaderMake/ShaderBlob.cpp)
  target_include_directories(NRD PUBLIC ${IROH_NRD_SRC}/Include PRIVATE ${IROH_NRD_SRC}/_Shaders ${IROH_NRD_MATHLIB_DIR} ${IROH_NRD_SHADERMAKE_DIR})
  # keep in sync with nrd-src/CMakeLists.txt (SPIRV_*REG_OFFSET, NRD_SUPPORTS_* options at their defaults + QUAD)
  set(_quad 0)
  if(IROH_NRD_QUAD)
    set(_quad 1)
  endif()
  target_compile_definitions(NRD PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS NRD_EMBEDS_SPIRV_SHADERS=1
    NRD_EMBEDS_DXIL_SHADERS=0 NRD_EMBEDS_DXBC_SHADERS=0 SPIRV_SREG_OFFSET=0 SPIRV_BREG_OFFSET=2 SPIRV_UREG_OFFSET=3 SPIRV_TREG_OFFSET=20
    NRD_SUPPORTS_VIEWPORT_OFFSET=0 NRD_SUPPORTS_CHECKERBOARD=1 NRD_SUPPORTS_HISTORY_CONFIDENCE=1 NRD_SUPPORTS_DISOCCLUSION_THRESHOLD_MIX=1
    NRD_SUPPORTS_ANTIFIREFLY=1 NRD_SUPPORTS_QUAD_INTRINSICS=${_quad} REBLUR_PERFORMANCE_MODE=0 NRD_STATIC_LIBRARY=1)
  # mingw: a bare abs(float) in MathLib binds ::abs(int) unless the C++ <cmath>/<stdlib.h> wrappers (using std::abs) are seen first
  target_compile_options(NRD PRIVATE -fvisibility=hidden -mssse3 -w "SHELL:-include cmath" "SHELL:-include stdlib.h")
  target_compile_features(NRD PUBLIC cxx_std_17)
else()
  add_subdirectory(${IROH_NRD_SRC} nrd EXCLUDE_FROM_ALL)
endif()
add_library(iroh_nrd SHARED src/iroh_nrd_jni.cpp)
if(WIN32)
  set_target_properties(iroh_nrd PROPERTIES PREFIX "" OUTPUT_NAME iroh_nrd)   # NativeLib.fileName: iroh_nrd.dll on Windows
else()
  set_target_properties(iroh_nrd PROPERTIES PREFIX "lib" OUTPUT_NAME iroh_nrd)
endif()
target_include_directories(iroh_nrd PRIVATE ${IROH_NRD_SRC}/Include ${IROH_JNI_INCLUDE})
target_link_libraries(iroh_nrd PRIVATE NRD)
target_compile_options(iroh_nrd PRIVATE -fvisibility=hidden)
if(WIN32)
  target_link_options(iroh_nrd PRIVATE -Wl,--no-insert-timestamp -static -static-libgcc -static-libstdc++ -Wl,--exclude-libs,ALL -Wl,--no-undefined)
else()
  target_link_options(iroh_nrd PRIVATE -static-libstdc++ -static-libgcc -Wl,--exclude-libs,ALL -Wl,--no-undefined)
endif()
